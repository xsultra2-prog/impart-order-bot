def load_json(file_path):
    if os.path.exists(file_path):
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            if file_path == USERS_FILE:
                return {}
            if file_path == DEBTS_FILE:
                return []
            return {"total_orders": 0, "cash_orders": 0, "debt_orders": 0, "total_sum": 0, "debts_list": []}
            
    if file_path == USERS_FILE:
        return {}
    if file_path == DEBTS_FILE:
        return []
    return {"total_orders": 0, "cash_orders": 0, "debt_orders": 0, "total_sum": 0, "debts_list": []}

def save_json(file_path, data):
    with open(file_path, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=4)

def extract_number(text):
    numbers = re.findall(r"\d+", text.replace(" ", "").replace(",", "."))
    if numbers:
        try:
            return float("".join(numbers)) if "." in "".join(numbers) else int("".join(numbers))
        except:
            return 0
    return 0

@dp.message(F.text.in_({"🛒 Buyurtma berish", "🛒 Буюртма бериш", "🛒 Сделать заказ", "🛒 Buyurtmani davom ettirish", "🛒 Буюртмани давом эттириш", "🛒 Продолжить заказ"}))
async def start_order(message: types.Message, state: FSMContext):
    user_id = message.from_user.id
    str_user_id = str(user_id)
    debts = load_json(DEBTS_FILE)
    today_str = datetime.now().strftime("%Y-%m-%d")

    overdue_debt = None
    for d in debts:
        if d.get("user_id") == user_id:
            due_date = d.get("date", "")
            if due_date and due_date < today_str:
                overdue_debt = d
                break

    if overdue_debt:
        username = message.from_user.username
        nickname_display = f"@{username}" if username else html.escape(message.from_user.full_name)
        warn_msg = get_text(user_id, "overdue", name=nickname_display)
        await message.answer(warn_msg, parse_mode="HTML")
        return

    users_data = load_json(USERS_FILE)
    if str_user_id in users_data:
        u_info = users_data[str_user_id]
        if u_info.get("receiver_name") and u_info.get("phone"):
            await state.update_data(
                receiver_name=u_info.get("receiver_name"),
                phone=u_info.get("phone")
            )
            await state.set_state(OrderState.waiting_for_order_text)
            await message.answer(
                get_text(user_id, "ask_order_text"),
                reply_markup=ReplyKeyboardRemove(),
                parse_mode="HTML"
            )
            return

    await state.set_state(OrderState.waiting_for_receiver_name)
    await message.answer(
        get_text(user_id, "ask_name"),
        reply_markup=ReplyKeyboardRemove(),
        parse_mode="HTML"
    )

@dp.message(OrderState.waiting_for_receiver_name, F.text)
async def process_receiver_name(message: types.Message, state: FSMContext):
    name = message.text.strip()
    if not name or name.isdigit():
        await message.answer("⚠️ Ism noto'g'ri! Qayta kiriting:", parse_mode="HTML")
        return
    
    username = message.from_user.username
    full_display_name = f"{name} (@{username})" if username else f"{name}"
    await state.update_data(receiver_name=full_display_name)
    
    await state.set_state(OrderState.waiting_for_receiver_phone)
    await message.answer(get_text(message.from_user.id, "ask_phone"), parse_mode="HTML")

@dp.message(OrderState.waiting_for_receiver_phone, F.text)
async def process_receiver_phone(message: types.Message, state: FSMContext):
    phone = message.text.strip()
    clean_phone = re.sub(r"[^\d]", "", phone)
    
    if clean_phone.startswith("998"):
        if len(clean_phone) != 12:
            await message.answer("⚠️ Telefon raqami noto'g'ri! Qayta kiriting:", parse_mode="HTML")
            return
    else:
        if len(clean_phone) != 9:
            await message.answer("⚠️ Telefon raqami noto'g'ri! Qayta kiriting:", parse_mode="HTML")
            return
        phone = "+998" + clean_phone

    await state.update_data(phone=phone)
    
    users_data = load_json(USERS_FILE)
    user_id_str = str(message.from_user.id)
    if user_id_str not in users_data:
        users_data[user_id_str] = {}
    users_data[user_id_str]["receiver_name"] = (await state.get_data()).get("receiver_name")
    users_data[user_id_str]["phone"] = phone
    save_json(USERS_FILE, users_data)

    await state.set_state(OrderState.waiting_for_order_text)
    await message.answer(get_text(message.from_user.id, "ask_order_text"), parse_mode="HTML")

    user_id = call.from_user.id

    if action == "new":

        full_bts_text = f"📍 <b>Manzil:</b> {decoded_addr}"
        await state.update_data(bts_location=full_bts_text)
        await call.message.delete()

async def finalize_debt_order_directly(message: types.Message, state: FSMContext):
    data = await state.get_data()
    user = message.from_user
    user_id = user.id
    days = int(data.get('debt_days', 1))
    amount_text = str(data.get('debt_amount_text', '66'))
    numeric_value = float(data.get('debt_amount_value', 66))
    
    now = datetime.now()
    due_date_str = (now + timedelta(days=days)).strftime("%Y-%m-%d")

    o_text = str(data.get('order_text', ''))
    debts = load_json(DEBTS_FILE)
    debt_id = int(now.timestamp() * 1000)
    debt_entry = {
        "id": debt_id,
        "user_id": user.id,
        "full_name": data.get('receiver_name', user.full_name),
        "phone": f"Qabul qiluvchi: {data.get('phone')}",
        "product": f"{o_text} (Summasi: {amount_text})",
        "amount_value": numeric_value,
        "date": due_date_str
    }
    debts.append(debt_entry)
    save_json(DEBTS_FILE, debts)

    r_name = html.escape(str(data.get('receiver_name', '')))
    p_phone = html.escape(str(data.get('phone', '')))
    bts_loc = html.escape(str(data.get('bts_location', '')))

    admin_text = (
        f"🛒 <b>YANGI BUYURTMA (Qarzga)</b>\n\n"
        f"👤 {r_name}\n📞 {p_phone}\n🆔 <code>{user.id}</code>\n"
        f"🏬 {bts_loc}\n📝 {o_text}\n💰 Summa: {amount_text}\n⏰ Muddat: {days} kun (Sana: {due_date_str})"
    )
    for admin_id in ADMINS:
        await bot.send_message(chat_id=admin_id, text=admin_text, reply_markup=get_admin_action_kb(user.id, "debt"), parse_mode="HTML")

    users_data = load_json(USERS_FILE)
    lang = users_data.get(str(user_id), {}).get("lang", "uz_lat")
    after_order_kb = InlineKeyboardMarkup(
        inline_keyboard=[
            [InlineKeyboardButton(text=LANG_DATA[lang]["btn_main_menu"], callback_data="after_main_menu")],
            [InlineKeyboardButton(text=LANG_DATA[lang]["btn_not_needed"], callback_data="after_not_needed")]
        ]
    )

    retry_reason = data.get('retry_reason_text')
    if retry_reason:
        msg_text = get_text(user_id, "order_sent_retry", reason=retry_reason)
    else:
        msg_text = get_text(user_id, "order_sent")

    await message.answer(msg_text, reply_markup=after_order_kb, parse_mode="HTML")
    await state.clear()
